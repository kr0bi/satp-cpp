\begin{frame}{Problema: contare il numero di elementi distinti}
  \begin{itemize}
    \item I \textbf{sistemi} moderni ricevono aggiornamenti in \textbf{modo continuo} e su \textbf{enormi} moli di \textbf{dati}.
    \item Il \textbf{problema} è: \textbf{stimare} quanti \textbf{elementi distinti} sono comparsi in un flusso.
    \item \textbf{Vincolo}: non poter conservare l'intero flusso in memoria.
    \item \textbf{Applicazioni}: \textit{monitoraggio di rete}, \textit{telemetria}, \textit{osservabilit\`a}, \textit{log}, \textit{ottimizzazione} di interrogazioni e \textit{sistemi distribuiti}.
  \end{itemize}

  \begin{block}{Perch\'e stimare?}
    Quando il volume dei dati cresce, il conteggio esatto diventa costoso in memoria e tempo, mentre la risposta deve arrivare con una bassa latenza.
  \end{block}
\end{frame}

\begin{frame}{Esempio introduttivo}
  \[
    S = \langle A, G, T, C, C, G, A \rangle
  \]

  \vspace{0.25cm}

  \[
    \text{elementi osservati} = 7
    \qquad
    \text{elementi distinti} = |\{A,C,G,T\}| = 4
  \]

  \vspace{0.25cm}

  \begin{itemize}
    \item Le ripetizioni non aumentano il numero di elementi distinti.
    \item Il valore cercato non \`e la lunghezza del flusso, ma il numero di valori diversi comparsi.
    \item Nei casi reali il flusso pu\`o essere molto grande o non avere una dimensione nota a priori.
  \end{itemize}

  \vspace{0.15cm}

  \centering
  \Large\(\text{domanda: quanti valori diversi ho visto finora?}\)
\end{frame}

\begin{frame}{Soluzione esatta naive}
  \[
    V \leftarrow V \cup \{x_i\}
    \qquad
    \text{risposta} = |V|
  \]

  \vspace{0.15cm}

  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.48\textwidth}
      \centering
      \begin{tabular}{c|l}
        \toprule
        Arriva & \(V\) \\
        \midrule
        \(A\) & \(\{A\}\) \\
        \(G\) & \(\{A,G\}\) \\
        \(T\) & \(\{A,G,T\}\) \\
        \(C\) & \(\{A,C,G,T\}\) \\
        \(C,G,A\) & invariato \\
        \bottomrule
      \end{tabular}
    \column{0.48\textwidth}
      \begin{itemize}
        \item Si mantiene l'insieme \(V\) dei valori gi\`a osservati.
        \item Ogni nuovo elemento aggiorna \(V\), se non era gi\`a presente.
        \item La risposta \`e esatta: \( |V| = 4 \).
      \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Limite della soluzione esatta}
  \[
    \text{memoria richiesta} \propto |V|
  \]

  \vspace{0.2cm}

  \begin{itemize}
    \item La soluzione naive deve conservare un rappresentante per ogni elemento distinto.
    \item Se il numero di distinti cresce, cresce anche lo stato mantenuto in memoria.
    \item Nei flussi continui il valore finale pu\`o non essere noto in anticipo.
    \item Spesso serve aggiornare la risposta con latenza bassa, senza analizzare tutto a posteriori.
  \end{itemize}

  \begin{alertblock}{Motivazione}
    Il problema non \`e definire cosa contare: \`e contarlo con memoria limitata.
  \end{alertblock}
\end{frame}

\begin{frame}{Idea dello sketch}
  \[
    x_i
    \longrightarrow
    \text{hash}
    \longrightarrow
    M
    \longrightarrow
    \widehat{F}_0
  \]

  \vspace{0.2cm}

  \[
    M \leftarrow \mathrm{Update}(M,x_i)
    \qquad
    \widehat{F}_0 \leftarrow \mathrm{Query}(M)
  \]

  \vspace{0.2cm}

  \begin{itemize}
    \item Lo sketch sostituisce l'insieme \(V\) con uno stato compatto \(M\).
    \item A ogni elemento lo stato viene aggiornato, senza conservare tutto il flusso.
    \item La risposta diventa una stima \(\widehat{F}_0\), non necessariamente il valore esatto.
    \item L'hash rende osservabili propriet\`a statistiche del flusso.
  \end{itemize}

  \begin{block}{Passaggio concettuale}
    Da ricordare quali elementi sono comparsi a mantenere abbastanza informazione per stimare quanti ne sono comparsi.
  \end{block}
\end{frame}

\begin{frame}{Caso distribuito}
  \[
    S_1 \longrightarrow K_1
    \qquad
    S_2 \longrightarrow K_2
    \qquad
    S_3 \longrightarrow K_3
  \]

  \vspace{0.2cm}

  \[
    \mathrm{Query}(K_1 \oplus K_2 \oplus K_3)
    \longrightarrow
    \widehat{F}_0(S_1 \cup S_2 \cup S_3)
  \]

  \vspace{0.2cm}

  \begin{itemize}
    \item Il flusso pu\`o essere diviso tra nodi o partizioni temporali.
    \item Ogni nodo mantiene solo il proprio sketch locale \(K_i\).
    \item Per stimare il numero globale di distinti, gli sketch locali devono essere uniti.
    \item Il merge dovrebbe comportarsi come se avessimo processato tutto il flusso in un solo sketch.
  \end{itemize}

  \begin{alertblock}{Punto critico}
    Il merge non combina solo stati: deve preservare la semantica con cui quegli stati sono stati costruiti.
  \end{alertblock}
\end{frame}

\begin{frame}{Domande del lavoro}
  \begin{itemize}
    \item Quando il merge di due sketch \`e equivalente al processamento centralizzato?
    \item Quali condizioni devono coincidere: algoritmo, hash, seed, precisione?
    \item Cosa succede se cambia la semantica dell'hash?
    \item Cosa succede se cambia solo la precisione dello sketch?
    \item Quali mismatch sono recuperabili e quali rendono il risultato privo di significato?
  \end{itemize}

  \begin{block}{Contributo}
    Classificare e misurare sperimentalmente i casi \texttt{valid}, \texttt{recoverable} e \texttt{invalid}.
  \end{block}
\end{frame}

\begin{frame}{Formalizzazione minima}
  \[
    S=\langle x_1,x_2,\dots,x_s\rangle,
    \qquad x_i\in\mathcal{U}
  \]

  \[
    f(a)=|\{i\mid x_i=a\}|
    \qquad
    \Fzero=|\{a\in\mathcal{U}\mid f(a)>0\}|
  \]

  \vspace{0.2cm}

  \begin{itemize}
    \item \(S\) \`e un flusso ordinato di elementi osservati.
    \item \(f(a)\) conta quante volte compare la chiave \(a\).
    \item \(\Fzero\) \`e la cardinalit\`a del supporto di \(f\), cio\`e il numero di elementi distinti.
    \item Nel lavoro si considera il modello \emph{insertion-only}: gli aggiornamenti aggiungono occorrenze.
  \end{itemize}
\end{frame}

\begin{frame}{Algoritmo di streaming}
  \[
    M \leftarrow \mathrm{Update}(M,x_i),
    \qquad
    \widehat{\Fzero}\leftarrow \mathrm{Query}(M)
  \]

  \vspace{0.2cm}

  \begin{itemize}
    \item Lo stato \(M\) deve essere piccolo rispetto alla dimensione del flusso.
    \item L'aggiornamento deve essere veloce, perch\'e avviene per ogni elemento.
    \item L'accuratezza viene letta come scostamento tra \(\widehat{\Fzero}\) e \(\Fzero\).
    \item L'obiettivo pratico \`e scambiare memoria per una stima controllata.
  \end{itemize}

  \begin{block}{Garanzia teorica tipica}
    Una stima \((\varepsilon,\delta)\)-approssimata richiede
    \(\Pr(|\widehat{\Fzero}-\Fzero|\le \varepsilon\Fzero)\ge 1-\delta\).
  \end{block}
\end{frame}

\begin{frame}{Linea evolutiva degli sketch}
  \scriptsize
  \begin{tabularx}{\textwidth}{>{\RaggedRight\arraybackslash}p{2.5cm} >{\RaggedRight\arraybackslash}p{2.7cm} >{\RaggedRight\arraybackslash}X}
    \toprule
    Metodo & Stato mantenuto & Miglioramento introdotto \\
    \midrule
    Probabilistic Counting & Una bitmap & Eventi rari dell'hash \\
    PCSA & Pi\`u bitmap virtuali & Mediazione su sottostrutture \\
    LogLog & Registri & Massimi di rango \\
    HyperLogLog & Registri & Media armonica e correzioni \\
    HyperLogLog++ & Sparso/denso & Bias correction e gestione pratica dei regimi piccoli \\
    \bottomrule
  \end{tabularx}

  \vspace{0.15cm}
  \small
  Idea comune: l'hash trasforma gli elementi in segnali pseudo-casuali; lo sketch osserva eventi rari per inferire la scala di \(\Fzero\).
\end{frame}

\begin{frame}{Probabilistic Counting}
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.48\textwidth}
      \[
        B=(b_1,\dots,b_L)
      \]
      \[
        \rho(y)=\min\{1+\nu_2(y),L\}
      \]
      \[
        b_{\rho(h(x))}\leftarrow 1
      \]

    \column{0.52\textwidth}
      \begin{itemize}
        \item Lo stato \`e una bitmap inizializzata a zero.
        \item \(\rho\) misura quanti zeri finali compaiono nell'hash, pi\`u uno.
        \item Eventi con \(\rho\) alto sono rari: vederli suggerisce molti distinti.
      \end{itemize}

      \[
        R(B)=\min\{r\ge 1:b_r=0\},
        \qquad
        \widehat{\Fzero}\approx \frac{2^{R(B)}}{\phi}
      \]
  \end{columns}

  \begin{alertblock}{Limite}
    Una sola bitmap \`e molto compatta, ma produce varianza elevata.
  \end{alertblock}
\end{frame}

\begin{frame}{LogLog}
  \[
    m=2^k,
    \qquad
    h(x)=\underbrace{j}_{\text{registro}}\ \underbrace{w}_{\text{rango}}
  \]

  \[
    M[j]\leftarrow \max\{M[j],\rho(w)\}
  \]

  \vspace{0.15cm}

  \begin{itemize}
    \item Il prefisso dell'hash sceglie uno dei \(m\) registri.
    \item Il suffisso produce il rango \(\rho(w)\).
    \item Ogni registro mantiene il massimo rango osservato.
    \item Se \(F_0\) cresce, i massimi tendono a crescere come \(\log_2(F_0/m)\).
  \end{itemize}

  \[
    A=\frac{1}{m}\sum_{j=0}^{m-1}M[j],
    \qquad
    \widehat{\Fzero}=\alpha_m\,m\,2^A
  \]
\end{frame}

\begin{frame}{Da LogLog a HyperLogLog}
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.47\textwidth}
      \begin{block}{Stato}
        \[
          M[j]\leftarrow \max\{M[j],\rho(w)\}
        \]
        HLL conserva la stessa struttura a registri di LogLog.
      \end{block}
    \column{0.53\textwidth}
      \begin{block}{Stimatore}
        \[
          Z=\sum_{j=0}^{m-1}2^{-M[j]},
          \qquad
          E=\alpha_m\frac{m^2}{Z}
        \]
        La media armonica riduce il peso dei registri estremi.
      \end{block}
  \end{columns}

  \vspace{0.15cm}
  \begin{itemize}
    \item Per cardinalit\`a piccole usa \emph{linear counting}: \(m\log(m/V_0)\).
    \item Nel regime centrale usa la stima armonica \(E\).
    \item Vicino al limite dello spazio hash applica una correzione di saturazione.
  \end{itemize}
\end{frame}

\begin{frame}{Da HyperLogLog a HyperLogLog++}
  \begin{itemize}
    \item HLL++ mantiene la logica a registri, ma usa hash a 64 bit.
    \item Per cardinalit\`a piccole usa una rappresentazione sparsa.
    \item Quando lo stato sparso non conviene pi\`u, passa alla rappresentazione densa.
    \item Corregge empiricamente il bias tramite tabelle dipendenti dalla precisione.
    \item Sceglie tra linear counting e stima corretta con soglie empiriche.
  \end{itemize}

  \[
    E_{\mathrm{corr}}=
    \begin{cases}
      E-\beta(E,k), & E\le 5m,\\
      E, & \text{altrimenti.}
    \end{cases}
  \]

  \begin{block}{Ruolo nella tesi}
    HLL++ \`e il riferimento pratico principale per gli esperimenti di streaming e di merge.
  \end{block}
\end{frame}

\begin{frame}{Merge e compatibilit\`a semantica}
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.50\textwidth}
      \begin{block}{Bitmap}
        Per Probabilistic Counting e PCSA:
        \[
          B=B_1\lor B_2
        \]
      \end{block}
    \column{0.50\textwidth}
      \begin{block}{Registri}
        Per LogLog, HLL e HLL++:
        \[
          M[j]=\max\{M_1[j],M_2[j]\}
        \]
      \end{block}
  \end{columns}

  \vspace{0.15cm}
  \begin{alertblock}{Condizione}
    Il merge coincide con il processamento centralizzato solo se lo stato ha la stessa semantica: algoritmo, hash, seed e parametri compatibili.
  \end{alertblock}
\end{frame}

\begin{frame}{Classificazione dei casi di merge}
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.33\textwidth}
      \begin{block}{\texttt{valid}}
        Merge diretto.

        Stesso algoritmo, stessa hash, stesso seed, stessi parametri.
      \end{block}
    \column{0.33\textwidth}
      \begin{block}{\texttt{recoverable}}
        Merge corretto dopo normalizzazione.

        HLL++ con precisioni diverse e stessa semantica hash.
      \end{block}
    \column{0.33\textwidth}
      \begin{block}{\texttt{invalid}}
        Merge privo di significato nel modello.

        Hash family o seed diversi.
      \end{block}
  \end{columns}

  \vspace{0.15cm}
  \small
  Strategie operative: \texttt{direct}, \texttt{reject}, \texttt{reduce\_then\_merge};
  \texttt{unsafe\_naive\_merge} \`e solo un controllo negativo.
\end{frame}

\begin{frame}{Obiettivi progettuali dell'implementazione}
  \begin{itemize}
    \item Uniformare algoritmi diversi sotto un contratto comune.
    \item Separare la responsabilit\`a dello sketch dalla responsabilit\`a dell'hashing.
    \item Rendere tracciabili seed, parametri, dataset, modalit\`a e risultati.
    \item Tenere separati aggiornamento dello sketch, checkpoint, metriche e serializzazione.
  \end{itemize}

  \begin{block}{Principio guida}
    Il progetto non \`e solo una raccolta di algoritmi: \`e una piattaforma di confronto riproducibile.
  \end{block}
\end{frame}

\begin{frame}{Disegno sperimentale}
  \scriptsize
  \begin{itemize}
    \item Ambiente di test: AMD Ryzen 7 9700X, 32 GB DDR5, Clang 22.1.1, CMake 4.3.0.
    \item Tre modalit\`a di valutazione:
    \begin{itemize}
      \item \textbf{streaming}: andamento della stima lungo i prefissi;
      \item \textbf{merge omogeneo}: confronto tra merge e riferimento seriale;
      \item \textbf{merge eterogeneo}: casi \texttt{valid}, \texttt{recoverable}, \texttt{invalid}.
    \end{itemize}
    \item Due viste complementari in streaming:

\begin{frame}{Sintesi dei risultati e contributi}
  \begin{columns}[T,totalwidth=\textwidth]
    \column{0.50\textwidth}
      \begin{block}{Risultati empirici}
        \begin{itemize}
          \item HLL++ \`e il metodo pi\`u stabile nel dominio osservato.
          \item Il merge omogeneo coincide col riferimento seriale.
          \item L'eterogeneit\`a dell'hash rompe la correttezza del merge.
          \item Il mismatch di precisione \`e recuperabile solo tramite normalizzazione esplicita.
        \end{itemize}
      \end{block}
    \column{0.50\textwidth}
      \begin{block}{Contributi del lavoro}
        \begin{itemize}
          \item Implementazione uniforme di pi\`u algoritmi count-distinct.
          \item Interfaccia comune per hash, dataset, metriche e serializzazione.
          \item Framework riutilizzabile per streaming e merge distribuito.
          \item Lessico operativo \texttt{valid}/\texttt{recoverable}/\texttt{invalid}.
        \end{itemize}
      \end{block}
  \end{columns}

  \begin{alertblock}{Take-away}
    Negli sketch count-distinct distribuiti, hash e parametri fanno parte della semantica dello sketch: senza compatibilit\`a semantica il merge non \`e un'operazione neutra.
  \end{alertblock}
\end{frame}
