            \begin{itemize}
              \item unbiased ($F_0 \approx \hat{F_0}$)
              \item update time $O(1)$
              \item small memory $\dots$
            \end{itemize}
        \end{itemize}

      \pause

      \column{0.38\textwidth}
        \begin{columns}[c,onlytextwidth]
          \column{0.5\textwidth}
            \underline{lower bound}\\
            $\Omega(\log N)$

          \column{0.5\textwidth}
            \underline{trivial upper bound}\\
            $O(m \cdot \log N)$
        \end{columns}

        \vspace{0.8em}
        where: \\
        $\mathbf{m} = $ \# of distinct elements \\
        $\mathbf{N} = $ \# of elements in the universe

        \pause
        \[
          \Theta(\epsilon^{-2} + \log N)
        \]
    \end{columns}

  \begin{frame}{Compatibilit\`a semantica del merge}
    \small
    Il merge \`e corretto solo se le due sketch descrivono i dati con la \alert{stessa semantica}: algoritmo, parametri e hash devono essere compatibili.

    \vspace{0.8em}

    \begin{columns}[T,onlytextwidth]
      \column{0.32\textwidth}
        \begin{exampleblock}{\texttt{valid}}
          Stessa semantica su entrambe le sketch.

          \vspace{0.4em}
          \textbf{Azione:} merge diretto.
        \end{exampleblock}

      \column{0.32\textwidth}
        \begin{block}{\texttt{recoverable}}
          Non direttamente compatibile, ma normalizzabile.

          \vspace{0.4em}
          \textbf{Azione:} riduzione e merge.
        \end{block}

      \column{0.32\textwidth}
        \begin{alertblock}{\texttt{invalid}}
          Non esiste un merge semanticamente corretto.

          \vspace{0.4em}
          \textbf{Azione:} rifiuto.
        \end{alertblock}
    \end{columns}
  \end{frame}
